#!/usr/bin/env python3
"""PreToolUse hook for wrought-runner's gate children. DEFENCE IN DEPTH — NOT THE BOUNDARY.

Phase-1 evidence (build-evidence/runner/raw/07) says two things about hooks that both matter:

  1. They DO fire under `claude -p` on claude 2.1.238, and a `deny` decision IS enforced even
     when --allowedTools would otherwise have permitted the call. So this is worth having.
  2. A settings file that fails to parse is SILENTLY ignored under -p — rc=0, empty stderr, and
     the entire hook layer is simply gone. So this can never be the layer that is relied upon.

The load-bearing containment is elsewhere: the kernel scope (MemoryMax/RuntimeMaxSec), the
default-deny permission mode with a per-gate allowlist, and the env allowlist. This hook adds an
audit trail of every tool call plus a small deny-list for the handful of actions that are
catastrophic rather than merely wrong. `wrought-runner` json-validates the settings file that
installs this hook before every single launch, precisely because of (2).

stdin : the PreToolUse JSON payload (session_id, cwd, tool_name, tool_input, ...)
stdout: a PreToolUse hookSpecificOutput decision
"""

from __future__ import annotations

import json
import os
import re
import sys
from datetime import datetime, timezone
from pathlib import Path

CONFIG = os.environ.get("WROUGHT_RUNNER_CONFIG", "/etc/wrought/runner.conf")

# Catastrophic-action patterns. Deliberately SHORT: a long denylist invites the belief that the
# list is the boundary. Each entry is something no gate has any business doing, ever.
DENY = [
    (re.compile(r"orchestrator\.db"),
     "the orchestrator store is byte-frozen for the whole batch (EXECUTOR-RAILS S2)"),
    (re.compile(r"\bsystemctl\b.*\b(start|stop|restart|enable|disable|daemon-reload)\b.*wrought-"),
     "wrought-* units are hands-off (EXECUTOR-RAILS S1)"),
    (re.compile(r"\bnft\b.*\bflush\b.*\bruleset\b"),
     "never `nft flush ruleset` (CLAUDE.md hard rule)"),
    (re.compile(r"--dangerously-skip-permissions|--allow-dangerously-skip-permissions"),
     "a gate may not re-launch itself without permission checks"),
    (re.compile(r"\bcredstore\.encrypted\b"),
     "sealed credentials are never read by a gate directly (docs/06 S14.4)"),
    (re.compile(r"\brm\b[^|;]*\s-\w*[rR]\w*f|\brm\b[^|;]*\s-\w*f\w*[rR]"),
     "recursive-force delete: deletes are enumerated, never globbed (EXECUTOR-RAILS S3)"),
]


def log_path() -> Path | None:
    try:
        cfg = json.loads(Path(CONFIG).read_text())
        p = Path(cfg["state_dir"]) / "hook-audit.jsonl"
        p.parent.mkdir(parents=True, exist_ok=True)
        return p
    except Exception:
        return None


def decide(payload: dict) -> tuple[str, str]:
    """Deny the catastrophic; DEFER on everything else.

    Deferring — not allowing — is the point. This hook only ever subtracts. Anything it does not
    deny is left to the permission mode and the gate's declared allowlist to decide, so the hook
    can never widen a gate's surface, only narrow it.
    """
    blob = json.dumps(payload.get("tool_input", {}))
    for pat, why in DENY:
        if pat.search(blob):
            return "deny", f"wrought-runner-hook: {why}"
    return "", ""


def main() -> int:
    raw = sys.stdin.read()
    try:
        payload = json.loads(raw)
    except Exception:
        # Fail CLOSED: an unparseable payload is not a reason to wave a tool call through.
        print(json.dumps({"hookSpecificOutput": {
            "hookEventName": "PreToolUse", "permissionDecision": "deny",
            "permissionDecisionReason": "wrought-runner-hook: unparseable PreToolUse payload"}}))
        return 0

    decision, reason = decide(payload)
    lp = log_path()
    if lp:
        with lp.open("a") as fh:
            fh.write(json.dumps({
                "at": datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),
                "session_id": payload.get("session_id"),
                "cwd": payload.get("cwd"),
                "permission_mode": payload.get("permission_mode"),
                "tool": payload.get("tool_name"),
                "input": payload.get("tool_input"),
                "decision": decision or "(defer)",
                "reason": reason,
            }, sort_keys=True) + "\n")

    if decision == "deny":
        print(json.dumps({"hookSpecificOutput": {
            "hookEventName": "PreToolUse", "permissionDecision": "deny",
            "permissionDecisionReason": reason}}))
    else:
        # Emit nothing decisive: the permission mode and the gate's own allowlist decide.
        print(json.dumps({"hookSpecificOutput": {"hookEventName": "PreToolUse"}}))
    return 0


if __name__ == "__main__":
    sys.exit(main())
